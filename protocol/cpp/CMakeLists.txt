cmake_minimum_required(VERSION 3.25)

# X4MP wire protocol, C++ side (task M0-07).
#
# Standalone project that M0-08's mod can also pull in with add_subdirectory(protocol/cpp). Targets:
#   x4mp::proto  FlatBuffers headers generated from protocol/schema/*.fbs by the pinned flatc
#                (tools/flatc, 25.2.10), plus the FlatBuffers runtime.
#   x4mp::wire   frame/datagram headers, Replication codec, quantisation, HMAC interface, and the message
#                catalog + Verifier (include/x4mp/wire.h, registry.h).
# Tests (Catch2, ctest) build only when this is the top-level project, or with -DX4MP_PROTOCOL_BUILD_TESTS=ON.
#
# The FlatBuffers runtime must be EXACTLY 25.2.10 (generated headers static_assert the version). Standalone,
# vcpkg.json pins it with an override; as a subproject, the parent must provide the same version.

project(x4mp_protocol VERSION 0.1.0 LANGUAGES CXX)

option(X4MP_PROTOCOL_BUILD_TESTS "Build the protocol tests" ${PROJECT_IS_TOP_LEVEL})

set(X4MP_FLATBUFFERS_VERSION "25.2.10")
get_filename_component(X4MP_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(X4MP_SCHEMA_DIR "${X4MP_REPO_ROOT}/protocol/schema")
set(X4MP_TESTDATA_DIR "${X4MP_REPO_ROOT}/protocol/testdata" CACHE PATH "Golden vectors directory (index.json + *.bin)")

# ---------------------------------------------------------------------------------------------
# flatc (pinned binary, fetched by tools/flatc/fetch-flatc.ps1 or .sh)
# ---------------------------------------------------------------------------------------------
if(CMAKE_HOST_WIN32)
  set(_x4mp_flatc_default "${X4MP_REPO_ROOT}/tools/flatc/bin/flatc.exe")
else()
  set(_x4mp_flatc_default "${X4MP_REPO_ROOT}/tools/flatc/bin/flatc")
endif()
set(X4MP_FLATC "${_x4mp_flatc_default}" CACHE FILEPATH "Pinned flatc ${X4MP_FLATBUFFERS_VERSION}")
if(NOT EXISTS "${X4MP_FLATC}")
  message(FATAL_ERROR "flatc not found at '${X4MP_FLATC}'. Run 'powershell -NoProfile -File tools/flatc/fetch-flatc.ps1' "
                      "(Windows) or 'tools/flatc/fetch-flatc.sh' (Linux) once, then configure again.")
endif()
execute_process(COMMAND "${X4MP_FLATC}" --version OUTPUT_VARIABLE _x4mp_flatc_version OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT _x4mp_flatc_version MATCHES "${X4MP_FLATBUFFERS_VERSION}$")
  message(FATAL_ERROR "flatc must be ${X4MP_FLATBUFFERS_VERSION}, got '${_x4mp_flatc_version}' (${X4MP_FLATC})")
endif()

# ---------------------------------------------------------------------------------------------
# FlatBuffers runtime: exactly 25.2.10
# ---------------------------------------------------------------------------------------------
find_package(Flatbuffers CONFIG REQUIRED)
if(NOT Flatbuffers_VERSION VERSION_EQUAL X4MP_FLATBUFFERS_VERSION)
  message(FATAL_ERROR "FlatBuffers runtime ${Flatbuffers_VERSION} found, ${X4MP_FLATBUFFERS_VERSION} required "
                      "(generated headers static_assert the version; see protocol/cpp/vcpkg.json overrides).")
endif()

# ---------------------------------------------------------------------------------------------
# Code generation: flatc --cpp --scoped-enums into the build dir
# ---------------------------------------------------------------------------------------------
file(GLOB X4MP_SCHEMAS CONFIGURE_DEPENDS "${X4MP_SCHEMA_DIR}/*.fbs")
set(X4MP_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${X4MP_GEN_DIR}")

set(X4MP_GENERATED_HEADERS "")
set(_x4mp_all_includes "// Generated by CMake: pulls in every schema so one .bfbs describes all tables.\n")
foreach(_schema IN LISTS X4MP_SCHEMAS)
  get_filename_component(_stem "${_schema}" NAME_WE)
  list(APPEND X4MP_GENERATED_HEADERS "${X4MP_GEN_DIR}/${_stem}_generated.h")
  if(NOT _stem STREQUAL "manifest") # manifest.fbs has a root_type; it is not part of the wire catalog
    string(APPEND _x4mp_all_includes "include \"${_stem}.fbs\";\n")
  endif()
endforeach()
file(WRITE "${X4MP_GEN_DIR}/x4mp_all.fbs" "${_x4mp_all_includes}")

add_custom_command(
  OUTPUT ${X4MP_GENERATED_HEADERS}
  COMMAND "${X4MP_FLATC}" --cpp --scoped-enums -o "${X4MP_GEN_DIR}" -I "${X4MP_SCHEMA_DIR}" ${X4MP_SCHEMAS}
  DEPENDS ${X4MP_SCHEMAS} "${X4MP_FLATC}"
  COMMENT "flatc ${X4MP_FLATBUFFERS_VERSION}: generating C++ from protocol/schema"
  VERBATIM)

add_library(x4mp_proto INTERFACE)
add_library(x4mp::proto ALIAS x4mp_proto)
# SYSTEM: generated code is not ours to warn about.
target_include_directories(x4mp_proto SYSTEM INTERFACE "${X4MP_GEN_DIR}")
target_link_libraries(x4mp_proto INTERFACE flatbuffers::flatbuffers)
add_custom_target(x4mp_proto_generate DEPENDS ${X4MP_GENERATED_HEADERS})
add_dependencies(x4mp_proto x4mp_proto_generate)

# ---------------------------------------------------------------------------------------------
# x4mp::wire
# ---------------------------------------------------------------------------------------------
add_library(x4mp_wire STATIC src/registry.cpp)
add_library(x4mp::wire ALIAS x4mp_wire)
target_sources(x4mp_wire PUBLIC FILE_SET HEADERS BASE_DIRS include FILES include/x4mp/wire.h include/x4mp/registry.h)
target_compile_features(x4mp_wire PUBLIC cxx_std_23)
target_link_libraries(x4mp_wire PUBLIC x4mp::proto)
set_target_properties(x4mp_wire PROPERTIES CXX_EXTENSIONS OFF)

# Our code: /W4 /permissive- and warnings are errors. Third-party and generated headers are SYSTEM
# includes (/external:W0), so they never fail the build.
function(x4mp_set_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /WX /external:W0 /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
x4mp_set_warnings(x4mp_wire)

# ---------------------------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------------------------
if(X4MP_PROTOCOL_BUILD_TESTS)
  enable_testing()
  find_package(Catch2 3 CONFIG REQUIRED)
  find_package(nlohmann_json CONFIG REQUIRED)
  include(Catch)

  # One .bfbs describing every table: the tests walk decoded messages with FlatBuffers reflection.
  set(X4MP_BFBS "${X4MP_GEN_DIR}/x4mp_all.bfbs")
  add_custom_command(
    OUTPUT "${X4MP_BFBS}"
    COMMAND "${X4MP_FLATC}" -b --schema -o "${X4MP_GEN_DIR}" -I "${X4MP_SCHEMA_DIR}" "${X4MP_GEN_DIR}/x4mp_all.fbs"
    DEPENDS ${X4MP_SCHEMAS} "${X4MP_GEN_DIR}/x4mp_all.fbs" "${X4MP_FLATC}"
    COMMENT "flatc ${X4MP_FLATBUFFERS_VERSION}: generating binary schema for reflection"
    VERBATIM)

  add_executable(x4mp_protocol_tests
    tests/field_dump.cpp
    tests/test_golden.cpp
    tests/test_wire.cpp
    "${X4MP_BFBS}")
  target_link_libraries(x4mp_protocol_tests PRIVATE x4mp::wire Catch2::Catch2WithMain nlohmann_json::nlohmann_json)
  target_compile_definitions(x4mp_protocol_tests PRIVATE
    X4MP_TESTDATA_DIR="${X4MP_TESTDATA_DIR}"
    X4MP_BFBS_PATH="${X4MP_BFBS}")
  x4mp_set_warnings(x4mp_protocol_tests)
  catch_discover_tests(x4mp_protocol_tests WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" PROPERTIES TIMEOUT 60)
endif()
